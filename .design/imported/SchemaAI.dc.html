-- The orders of one customer, newest first,
-- with the number of lines of each order.
select o.order_id,
       o.created_at,
       o.status,
       o.total,
       count(l.order_line_id) as line_count
from orders o
left join order_line l on l.order_id = o.order_id
where o.customer_id = :customer_id
group by o.order_id
order by o.created_at desc
limit 100;
